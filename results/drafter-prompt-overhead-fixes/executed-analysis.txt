#!/usr/bin/env bash
set -u
cd /home/philip/.t3/worktrees/BitDraft/t3-fc0c881c
export BINARY_NUMERICAL_PYTHON=/home/philip/BitDraft/.venv/numerical-clean/bin/python
while [ $# -ge 3 ]; do
  cmd=$1; spec=$2; out=$3; shift 3
  while true; do
    log=$(mktemp)
    uv run --locked --no-sync --offline python -B -m benchmarks "$cmd" "$spec" "$out" > "$log" 2>&1
    code=$?
    if [ $code -ne 0 ] && grep -q -E "GPU lease|compute processes" "$log" && [ ! -e "$out" ]; then
      rm -f "$log"; sleep 30; continue
    fi
    echo "$(date +%H:%M:%S) $cmd $out exit $code"; tail -3 "$log"; rm -f "$log"
    break
  done
done
===== screen_check.py
import json
import sys
from pathlib import Path

a, b = Path(sys.argv[1]), Path(sys.argv[2])
out = {}
for arm in ("target", "baseline", "candidate"):
    cells_a = sorted(p for p in a.iterdir() if p.is_dir() and p.name.endswith("-" + arm))
    cells_b = sorted(p for p in b.iterdir() if p.is_dir() and p.name.endswith("-" + arm))
    assert [p.name for p in cells_a] == [p.name for p in cells_b]
    n = same = 0
    prompt = {"a": 0.0, "b": 0.0}
    pred = {"a": 0.0, "b": 0.0}
    acc = {"a": [0, 0], "b": [0, 0]}
    for ca, cb in zip(cells_a, cells_b):
        for ra in sorted(ca.glob("*-response.json")):
            rb = cb / ra.name
            x, y = json.loads(ra.read_text()), json.loads(rb.read_text())
            n += 1
            tx, ty = x["timings"], y["timings"]
            ok = (
                x.get("tokens") == y.get("tokens")
                and x.get("content") == y.get("content")
                and x.get("stop_type") == y.get("stop_type")
                and tx.get("draft_n") == ty.get("draft_n")
                and tx.get("draft_n_accepted") == ty.get("draft_n_accepted")
                and tx["prompt_n"] == ty["prompt_n"]
                and tx["predicted_n"] == ty["predicted_n"]
            )
            same += ok
            prompt["a"] += tx["prompt_ms"]
            prompt["b"] += ty["prompt_ms"]
            pred["a"] += tx["predicted_ms"]
            pred["b"] += ty["predicted_ms"]
            acc["a"][0] += tx.get("draft_n") or 0
            acc["a"][1] += tx.get("draft_n_accepted") or 0
            acc["b"][0] += ty.get("draft_n") or 0
            acc["b"][1] += ty.get("draft_n_accepted") or 0
    out[arm] = {
        "responses": n,
        "identical": same,
        "prompt_seconds": {k: round(v / 1000, 3) for k, v in prompt.items()},
        "prediction_seconds": {k: round(v / 1000, 3) for k, v in pred.items()},
        "proposed_accepted": acc,
    }
print(json.dumps(out, indent=1))
===== live_check.py
import json
import sys
from pathlib import Path

a, b = Path(sys.argv[1]), Path(sys.argv[2])
rows_a = json.loads((a / "progress.json").read_text())["rows"]
rows_b = json.loads((b / "progress.json").read_text())["rows"]
sa = json.loads((a / "result.json").read_text())["summary"]
sb = json.loads((b / "result.json").read_text())["summary"]
out = {}
for arm in ("target", "baseline", "candidate"):
    ra = {r["task"]: r for r in rows_a if r["arm"] == arm}
    rb = {r["task"]: r for r in rows_b if r["arm"] == arm}
    assert ra.keys() == rb.keys()
    tasks_same = calls = calls_same = 0
    for task in ra:
        x, y = ra[task], rb[task]
        keys = ("state_sha256", "termination", "turns", "tokens", "completed", "native")
        tasks_same += all(x[k] == y[k] for k in keys)
        ta = json.loads((a / ("0-" + arm) / task / "transcript.json").read_text())
        tb = json.loads((b / ("0-" + arm) / task / "transcript.json").read_text())
        assert len(ta["responses"]) == len(tb["responses"]) or True
        for p, q in zip(ta["responses"], tb["responses"]):
            calls += 1
            tp, tq = p.get("timings", {}), q.get("timings", {})
            calls_same += (
                p.get("tokens") == q.get("tokens")
                and p.get("content") == q.get("content")
                and tp.get("draft_n") == tq.get("draft_n")
                and tp.get("draft_n_accepted") == tq.get("draft_n_accepted")
                and tp.get("prompt_n") == tq.get("prompt_n")
                and tp.get("cache_n") == tq.get("cache_n")
            )
        calls += abs(len(ta["responses"]) - len(tb["responses"]))
        calls_same += 0
    keys = ("model_seconds", "server_prompt_seconds", "server_prediction_seconds", "elapsed_seconds", "completed", "accepted", "proposed", "tokens")
    out[arm] = {
        "tasks": len(ra),
        "tasks_identical": tasks_same,
        "calls": calls,
        "calls_identical": calls_same,
        "v5": {k: sa[arm][k] for k in keys},
        "v6": {k: sb[arm][k] for k in keys},
    }
print(json.dumps(out, indent=1))
