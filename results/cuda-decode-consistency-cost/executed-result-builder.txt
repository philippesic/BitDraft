def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def arm(name: str) -> dict:
    parts = name.split("-")
    depth = next((int(p[1:]) for p in parts if p.startswith("d") and p[1:].isdigit()), None)
    return {"arm": parts[0], "mode": "consistent" if parts[1] == "c" else "normal", "depth": depth}


def compact_divergence(div: dict, ref: str) -> dict:
    margins = []
    second = 0
    for item in div["differences"]:
        m = item.get("target_" + ref + "_margin")
        if m:
            margins.append(round(m["margin_logit"], 5))
            second += m["top2"] == item["token"]
    return {
        "requests_differing": div["requests_differing"],
        "requests": div["requests"],
        "first_differences_per_1000_tokens": round(div["first_differences_per_1000_tokens"], 3),
        "first_difference_positions": [d["first_difference"] for d in div["differences"]],
        "reference_logit_margins_at_first_difference": sorted(margins),
        "alternative_was_reference_second_choice": second,
        "margins_observed": len(margins),
    }


def runs(analysis: dict, batch_of: dict) -> list[dict]:
    rows = []
    for name, v in sorted(analysis.items()):
        row = {"run": name, "batch": batch_of.get(name), **arm(name), **v["summary"]}
        if name.startswith("target") or "-d2-" in name or name.endswith("-d4-r0"):
            row["length_bands"] = v["length_bands"]
        st = v.get("drafter_statistics")
        if st:
            row["drafter_draft_ms_per_round"] = round(st["drafter_draft_ms"] / st["rounds"], 4)
            row["reach_position"] = st["reach_position"]
            row["conditional_position"] = st["conditional_position"]
        for ref in ("c", "n"):
            if "versus_target_" + ref in v:
                row["versus_target_" + ("consistent" if ref == "c" else "normal")] = compact_divergence(
                    v["versus_target_" + ref], ref
                )
        if "verifier_trace" in v:
            row["verifier_trace"] = {k: v["verifier_trace"][k] for k in ("status", "rounds", "proposed", "accepted")}
        rows.append(row)
    return rows
