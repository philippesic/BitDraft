def sha(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1 << 24), b""):
            digest.update(block)
    return digest.hexdigest()


def parse(name: str) -> dict:

        "depth": int(parts[2][1:]) if len(parts) > 3 else None,
        "repetition": int(parts[-1][1:]),
    }

def paired_acceptance(root: Path, arm: str, depth: int, rep: str = "r0") -> dict:
    c = {r["task"]: r for r in analyze.load(root / f"{arm}-c-d{depth}-{rep}")}
    n = {r["task"]: r for r in analyze.load(root / f"{arm}-n-d{depth}-{rep}")}
    same = [t for t in c if c[t]["tokens"] == n[t]["tokens"]]
    return {
        "arm": arm,
        "depth": depth,
        "requests": len(c),
        "identical_output_requests": len(same),
        "accepted_consistent": sum(c[t]["draft_n_accepted"] for t in same),
        "accepted_normal": sum(n[t]["draft_n_accepted"] for t in same),
        "proposed_consistent": sum(c[t]["draft_n"] for t in same),
        "proposed_normal": sum(n[t]["draft_n"] for t in same),
        "requests_with_equal_accepted": sum(c[t]["draft_n_accepted"] == n[t]["draft_n_accepted"] for t in same),
    }
