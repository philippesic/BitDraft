def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def load(path):
    return json.loads(Path(path).read_text())

def rounded(value):
    if isinstance(value, float):
        return round(value, 4)
    if isinstance(value, dict):
        return {k: rounded(v) for k, v in value.items()}
    if isinstance(value, list):
        return [rounded(v) for v in value]
    return value

    match = re.search(r"end (\S+) exit=(\d+) seconds=([\d.]+)", line)
    if match:
        seconds[match.group(1)] = {"exit": int(match.group(2)), "seconds": round(float(match.group(3)), 1)}

def rows(run, arm):
    return {r["task"]: r for r in load(P / "runs" / run / "result.json")["rows"] if r["arm"] == arm and r["rep"] == 0}

def counts(run, arm):
    return {t: [r["native"]["accepted"], r["native"]["proposed"], r["tokens"]] for t, r in rows(run, arm).items()}

def transcripts(root, rep, arm):
    return {p.parent.name: load(p)["responses"] for p in (root / f"{rep}-{arm}").glob("*/transcript.json")}

live = {}
EARLIER_NVFP4 = {"runtime-v5 depth 2, saved-control source": REG.parent.parent / "qualification/lean-checks/nvfp4-controls-suite-01/nvfp4-live"}
def compare_target(run, earlier_map):
    target = transcripts(P / "runs" / run, 0, "target")
    for label, earlier in earlier_map.items():
        old = transcripts(earlier, 0, "target")
        calls = same_tokens = same_counts = 0
        for task, responses in target.items():
            for a, b in zip(responses, old[task]):
                calls += 1
                same_tokens += a["tokens"] == b["tokens"]
                same_counts += (a["timings"]["cache_n"], a["tokens_evaluated"]) == (b["timings"]["cache_n"], b["tokens_evaluated"])
        live[f"{run} target vs {label}"] = {"calls": calls, "identical_tokens": same_tokens, "identical_cached_and_evaluated": same_counts}

def compare_arm(target_run, run, arm):
    target = transcripts(P / "runs" / target_run, 0, "target")
    other = transcripts(P / "runs" / run, 0, arm)
    calls = differing_cache = 0
    events = []
    for task, responses in sorted(target.items()):
        for index, (a, b) in enumerate(zip(responses, other[task])):
            calls += 1
            if a["timings"]["cache_n"] != b["timings"]["cache_n"]:
                differing_cache += 1
                previous = other[task][index - 1]
                events.append({"task": task, "call": index, "target_cached": a["timings"]["cache_n"], "arm_cached": b["timings"]["cache_n"],
                               "previous_call_stop": previous["stop_type"], "previous_call_last_token": previous["tokens"][-1],
                               "previous_call_proposed": previous["timings"].get("draft_n"), "previous_call_accepted": previous["timings"].get("draft_n_accepted"),
                               "tokens_identical": a["tokens"] == b["tokens"]})
            if a["tokens"] != b["tokens"]:
                first = next((i for i, (x, y) in enumerate(zip(a["tokens"], b["tokens"])) if x != y), None)
                events.append({"task": task, "call": index, "first_differing_generated_token": first})
                break
    live[f"{run} {arm} vs {target_run} target"] = {"calls_compared": calls, "cached_prefix_differs": differing_cache, "events": events}
