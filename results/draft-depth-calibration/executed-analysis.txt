import json
import sys
from pathlib import Path

ROOT = Path(sys.argv[1])
REPO = Path(sys.argv[2])
fair = json.loads((REPO / "results/draft-depth-fairness/result.json").read_text())
sweep = json.loads((REPO / "results/cuda-decode-consistency-cost/result.json").read_text())


def measured_panel(arm: str) -> dict[int, dict[str, float]]:
    out: dict[int, list[dict]] = {}
    for run in fair["runs"]:
        if run["arm"] == arm and run["mode"] == "consistent":
            out.setdefault(run["depth"], []).append(run)
    return {
        depth: {
            "complete": sum(r["tokens_per_second"] for r in runs) / len(runs),
            "decode": sum(r["decode_tokens_per_second"] for r in runs) / len(runs),
            "emitted_per_round": 1 + runs[0]["accepted_per_round"],
            "ms_per_round": sum(r["decode_ms_per_round"] for r in runs) / len(runs),
        }
        for depth, runs in out.items()
    }


def measured_screen(arm: str) -> dict[int, dict[str, float]]:
    out: dict[int, list[dict]] = {}
    for run in sweep["screen_runs"]:
        if run["arm"] == arm and run["mode"] == "consistent" and run.get("depth"):
            out.setdefault(run["depth"], []).append(run)
    return {
        depth: {
            "complete": sum(r["tokens_per_second"] for r in runs) / len(runs),
            "decode": sum(r["decode_tokens_per_second"] for r in runs) / len(runs),
            "emitted_per_round": 1 + runs[0]["accepted_per_round"],
            "ms_per_round": sum(r["decode_ms_per_round"] for r in runs) / len(runs),
        }
        for depth, runs in out.items()
    }


def compare(name: str, measured: dict[int, dict[str, float]]) -> dict:
    path = ROOT / name / "calibration.json"
    if not path.exists():
        return {"missing": name}
    record = json.loads(path.read_text())
    predicted = {row["depth"]: row for row in record["predictions"]}
    depths = sorted(measured)
    anchor = depths[0]
    rows = []
    for depth in depths:
        m, p = measured[depth], predicted[depth]
        rows.append(
            {
                "depth": depth,
                "measured_complete": round(m["complete"], 1),
                "predicted_complete": round(p["complete_tokens_per_second"], 1),
                "measured_decode": round(m["decode"], 1),
                "predicted_decode": round(p["decode_tokens_per_second"], 1),
                "measured_complete_vs_anchor": round(m["complete"] / measured[anchor]["complete"], 4),
                "predicted_complete_vs_anchor": round(
                    p["complete_tokens_per_second"] / predicted[anchor]["complete_tokens_per_second"], 4
                ),
                "measured_decode_vs_anchor": round(m["decode"] / measured[anchor]["decode"], 4),
                "predicted_decode_vs_anchor": round(
                    p["decode_tokens_per_second"] / predicted[anchor]["decode_tokens_per_second"], 4
                ),
                "measured_emitted_per_round": round(m["emitted_per_round"], 3),
                "predicted_emitted_per_round": round(p["tokens_per_round"], 3),
                "measured_ms_per_round": round(m["ms_per_round"], 3),
                "predicted_ms_per_round": round(p["ms_per_round"], 3),
            }
        )
    best_measured = max(depths, key=lambda d: measured[d]["complete"])
    best_predicted_among = max(depths, key=lambda d: predicted[d]["complete_tokens_per_second"])
    return {
        "calibration": name,
        "anchor_depth": anchor,
        "rows": rows,
        "measured_best_among_measured_depths": best_measured,
        "predicted_best_among_measured_depths": best_predicted_among,
        "recommendation": record["recommendation"],
        "elapsed_seconds": round(record["elapsed_seconds"], 1),
        "survival_seconds": round(record["survival_seconds"], 1),
        "cost_seconds": round(sum(c["seconds"] for c in record["costs"]), 1),
        "survival_ms_per_round_vs_cost_session": round(
            record["survival"]["ms_per_round"]
            / next(c for c in record["costs"] if c["depth"] == record["settings"]["max_depth"])["ms_per_round"],
            4,
        ),
        "by_category": {k: {"depth": v["depth"], "near_best": v["near_best"], "drafted_rounds": v["drafted_rounds"]} for k, v in record["by_category"].items()},
    }


report = [
    compare("panel-baseline", measured_panel("baseline")),
    compare("panel-baseline-r1", measured_panel("baseline")),
    compare("panel-he4", measured_panel("he4")),
    compare("panel-armd", measured_panel("armd")),
    compare("screen-baseline", measured_screen("baseline")),
    compare("screen-he4", measured_screen("he4")),
]
json.dump(report, sys.stdout, indent=1)

ARMS=('baseline','he4','armd','baseline-r1')
out={}
for arm in ARMS:
    cal=json.load(open(root/f'panel-{arm}/calibration.json'))
    subset=set(cal['settings']['tasks'])
    pred={p['depth']:p for p in cal['predictions']}
    res={}
    for depth in (2,3):
        runs=[x for x in r['runs'] if x['arm']==arm.split('-')[0] and x['mode']=='consistent' and x['depth']==depth]
        vals=[]
        for x in runs:
            rows=[row for i,row in enumerate(x['requests_rows']) if ids[row[0]] in subset]
            tok=sum(w[1] for w in rows); n=len(rows); wall=sum(w[3] for w in rows); pm=sum(w[5] for w in rows); acc=sum(w[7] for w in rows)
            vals.append(dict(complete=tok/wall, decode=(tok-n)/(pm/1000), emitted=(tok-n)/(tok-acc-n), ms=pm/(tok-acc-n)))
        m={k:sum(v[k] for v in vals)/len(vals) for k in vals[0]}
        res[depth]={'subset_measured':{k:round(v,3) for k,v in m.items()},'predicted':{'complete':round(pred[depth]['complete_tokens_per_second'],3),'decode':round(pred[depth]['decode_tokens_per_second'],3),'emitted':round(pred[depth]['tokens_per_round'],3),'ms':round(pred[depth]['ms_per_round'],3)},'runs':len(vals)}
    out[arm]=res
json.dump(out,sys.stdout,indent=1)
